<script>
/* =========================================================
   Canva Modal (IMG + VIDEO)
   - Portals modal to <body> to avoid Hydejack transform stacking issues
   - Locks scroll
   - Sound toggle shows STATE (Sound: On/Off), not reversed
   - Sound only available when slide has data-has-audio="true"
   ========================================================= */
let canvaSlideIndex = 1;
let canvaSoundEnabled = false;

function openModal_canva() {
  const modal = document.getElementById("canva_modal");
  if (!modal) return;

  modal.style.display = "block";
  modal.setAttribute("aria-hidden", "false");

  document.documentElement.classList.add("canva-modal-open");
  document.body.classList.add("canva-modal-open");
}

function closeModal_canva() {
  const modal = document.getElementById("canva_modal");
  if (!modal) return;

  modal.style.display = "none";
  modal.setAttribute("aria-hidden", "true");

  document.documentElement.classList.remove("canva-modal-open");
  document.body.classList.remove("canva-modal-open");

  pauseAllModalVideos();
}

function plusSlides_canva(n) {
  showSlides_canva(canvaSlideIndex += n);
}

function currentSlide_canva(n) {
  showSlides_canva(canvaSlideIndex = n);
}

function pauseAllModalVideos(){
  const modal = document.getElementById("canva_modal");
  if (!modal) return;
  modal.querySelectorAll("video").forEach(v => {
    try { v.pause(); } catch(e){}
  });
}

function applySoundToVideo(v){
  if (!v) return;
  v.muted = !canvaSoundEnabled;
  v.volume = canvaSoundEnabled ? 1 : 0;
}

function updateSoundUI(activeSlide){
  const btn = document.getElementById("canva_sound_toggle");
  if (!btn) return;

  const hasAudio = activeSlide?.dataset?.hasAudio === "true";

  if (!hasAudio){
    btn.textContent = "Sound: N/A";
    btn.classList.add("is-disabled");
    btn.setAttribute("aria-pressed", "false");
    return;
  }

  btn.classList.remove("is-disabled");
  btn.textContent = canvaSoundEnabled ? "Sound: On" : "Sound: Off";
  btn.setAttribute("aria-pressed", canvaSoundEnabled ? "true" : "false");
}

function showSlides_canva(n) {
  const slides = Array.from(document.getElementsByClassName("mySlides_canva"));
  if (!slides.length) return;

  if (n > slides.length) canvaSlideIndex = 1;
  if (n < 1) canvaSlideIndex = slides.length;

  // hide all
  slides.forEach(s => s.classList.remove("is-active"));
  pauseAllModalVideos();

  const active = slides[canvaSlideIndex - 1];
  active.classList.add("is-active");

  // caption
  const cap = document.getElementById("caption_canva");
  if (cap) cap.textContent = active.dataset.caption || "";

  // counter
  const counter = document.getElementById("canva_counter");
  if (counter) counter.textContent = `${canvaSlideIndex} / ${slides.length}`;

  // sound UI
  updateSoundUI(active);

  // play video if present
  const v = active.querySelector("video");
  if (v){
    // only allow sound if slide has audio
    const hasAudio = active.dataset.hasAudio === "true";
    if (!hasAudio){
      v.muted = true;
      v.volume = 0;
    } else {
      applySoundToVideo(v);
    }
    v.play().catch(() => {});
  }
}

/* Sound toggle */
(() => {
  const btn = document.getElementById("canva_sound_toggle");
  if (!btn) return;

  btn.addEventListener("click", () => {
    if (btn.classList.contains("is-disabled")) return;

    canvaSoundEnabled = !canvaSoundEnabled;

    const active = document.querySelector(".mySlides_canva.is-active");
    updateSoundUI(active);

    const v = active ? active.querySelector("video") : null;
    if (v){
      applySoundToVideo(v);
      v.play().catch(() => {});
    }
  });
})();

/* Keyboard controls */
(() => {
  document.addEventListener("keydown", (e) => {
    const modal = document.getElementById("canva_modal");
    if (!modal || modal.style.display !== "block") return;

    if (e.key === "Escape") closeModal_canva();
    if (e.key === "ArrowLeft") plusSlides_canva(-1);
    if (e.key === "ArrowRight") plusSlides_canva(1);
  });
})();

/* Init */
document.addEventListener("DOMContentLoaded", () => {
  showSlides_canva(canvaSlideIndex);
});

/* ---------- Thumbnail videos: lazy-load + play only when visible ---------- */
(function(){
  document.addEventListener("DOMContentLoaded", () => {
    const vids = Array.from(document.querySelectorAll("video.js-thumbvid"));
    if (!vids.length) return;

    const reduce = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
    if (reduce) return;

    const io = new IntersectionObserver((entries) => {
      entries.forEach(entry => {
        const v = entry.target;
        const source = v.querySelector("source");
        if (!source) return;

        if (entry.isIntersecting) {
          // Set src only when visible (fast page load)
          if (!source.src) {
            source.src = source.dataset.src || "";
            v.load();
          }
          v.play().catch(() => {});
        } else {
          v.pause();
        }
      });
    }, { rootMargin: "160px 0px" });

    vids.forEach(v => io.observe(v));
  });
})();

/* ---------- Init first slide ---------- */
document.addEventListener("DOMContentLoaded", () => {
  showSlides_canva(canvaSlideIndex);
});
</script>
