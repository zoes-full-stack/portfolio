<script>
  function openModal_canva() {
    document.getElementById("canva_modal").style.display = "block";
  }

  function closeModal_canva() {
    document.getElementById("canva_modal").style.display = "none";
  }

  var canvaSlideIndex = 1;
  showSlides_canva(canvaSlideIndex);

  function plusSlides_canva(n) {
    showSlides_canva(canvaSlideIndex += n);
  }

  function currentSlide_canva(n) {
    showSlides_canva(canvaSlideIndex = n);
  }

  function showSlides_canva(n) {
    var i;
    var slides = document.getElementsByClassName("mySlides_canva");
    var dots = document.getElementsByClassName("demo_canva");
    var captionText = document.getElementById("caption_canva");

    if (n > slides.length) { canvaSlideIndex = 1; }
    if (n < 1) { canvaSlideIndex = slides.length; }

    for (i = 0; i < slides.length; i++) {
      slides[i].style.display = "none";
    }

    for (i = 0; i < dots.length; i++) {
      dots[i].className = dots[i].className.replace(" active", "");
    }

    slides[canvaSlideIndex - 1].style.display = "block";
    dots[canvaSlideIndex - 1].className += " active";

    captionText.innerHTML = dots[canvaSlideIndex - 1].alt || "";
  }

  // close when clicking outside modal
  var modal_canva = document.getElementById('canva_modal');
  window.addEventListener("click", function(event) {
    if (event.target == modal_canva) {
      modal_canva.style.display = "none";
    }
  });
</script>
